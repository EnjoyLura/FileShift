### 2.2 异步任务处理流程

```
用户上传文件
    │
    ▼
API 层校验（文件类型、大小、积分）
    │
    ▼
扣除用户积分（数据库事务）
    │
    ▼
文件保存至临时目录，写入 files 表
    │
    ▼
创建 task 记录（状态：pending），写入 tasks 表
    │
    ▼
将任务推入 BullMQ 队列（携带 taskId、fileId、参数）
    │
    ▼
返回 taskId 给前端
    │
    ▼（异步）
Worker 从队列取出任务
    │
    ▼
更新 task 状态为 processing
    │
    ▼
调用对应处理工具（FFmpeg/Sharp/LibreOffice 等）
    │
    ▼
处理完成：输出文件保存，更新 task 状态为 completed
    │
    ▼
前端通过轮询（或 WebSocket）获取任务状态，展示结果
    │
    ▼
用户下载结果文件
    │
    ▼（定时任务，每小时执行）
清理过期的临时文件和输出文件
```

### 2.3 微信登录 OAuth2.0 流程

```
用户点击"微信登录"
    │
    ▼
前端重定向至微信授权页面
https://open.weixin.qq.com/connect/qrconnect?appid=APPID&redirect_uri=REDIRECT_URI&response_type=code&scope=snsapi_login
    │
    ▼
用户扫码确认授权
    │
    ▼
微信回调至后端 /api/auth/wechat/callback?code=CODE
    │
    ▼
后端使用 code + appSecret 换取 access_token
https://api.weixin.qq.com/sns/oauth2/access_token
    │
    ▼
使用 access_token 获取用户信息
https://api.weixin.qq.com/sns/userinfo
    │
    ▼
根据 openid 查询用户表
    │
    ├── 用户已存在 → 生成 JWT Token，登录成功
    │
    └── 用户不存在 → 创建新用户，赠送50积分，生成 JWT Token
    │
    ▼
重定向至前端，携带 JWT Token
```

### 2.4 积分扣除流程（事务保证）

```
开始数据库事务
    │
    ▼
查询用户当前积分余额（SELECT ... FOR UPDATE 行锁）
    │
    ├── 积分不足 → 回滚事务，返回错误
    │
    └── 积分充足 → 继续
        │
        ▼
        扣除积分（UPDATE users SET points = points - N）
        │
        ▼
        写入积分流水记录（INSERT INTO points_transactions）
        │
        ▼
        创建任务记录
        │
        ▼
        提交事务
```

## 4. 数据库设计

### 4.1 ER 关系图

```
users (用户)
  ├── 1:N → files (文件)
  ├── 1:N → tasks (任务)
  ├── 1:N → points_transactions (积分流水)
  └── 1:N → sessions (会话)

tools (工具配置)
  └── 1:N → tasks (任务)

files (文件)
  └── N:1 → tasks (任务，输入/输出文件)
```

### 4.2 Prisma Schema 定义

```prisma
// prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============ 用户表 ============
model User {
  id            String    @id @default(uuid())
  nickname      String    @default("")
  avatar        String?
  email         String?   @unique
  emailVerified Boolean   @default(false)
  phone         String?   @unique
  phoneVerified Boolean   @default(false)
  passwordHash  String?   // 邮箱登录密码的 bcrypt 哈希
  wechatOpenid  String?   @unique @map("wechat_openid")
  wechatUnionid String?   @map("wechat_unionid")
  points        Int       @default(0)
  vipType       VipType?  @map("vip_type")
  vipExpiresAt  DateTime? @map("vip_expires_at")
  inviteCode    String    @unique @map("invite_code")
  invitedBy     String?   @map("invited_by")
  lastSignInAt  DateTime? @map("last_sign_in_at")
  status        UserStatus @default(ACTIVE)
  createdAt     DateTime  @default(now()) @map("created_at")
  updatedAt     DateTime  @updatedAt @map("updated_at")

  files             File[]
  tasks             Task[]
  pointsTransactions PointsTransaction[]

  @@map("users")
}

enum UserStatus {
  ACTIVE
  BANNED
  DELETED
}

enum VipType {
  MONTHLY
  QUARTERLY
  YEARLY
}

// ============ 文件表 ============
model File {
  id          String    @id @default(uuid())
  userId      String    @map("user_id")
  originalName String   @map("original_name")
  storedName  String    @unique @map("stored_name") // UUID + 扩展名
  mimeType    String    @map("mime_type")
  size        Int       // 文件大小（字节）
  storagePath String    @map("storage_path") // 存储路径
  fileType    FileType  @map("file_type")
  createdAt   DateTime  @default(now()) @map("created_at")
  expiresAt   DateTime? @map("expires_at") // 过期清理时间

  user        User      @relation(fields: [userId], references: [id])
  inputTasks  Task[]    @relation("InputFile")
  outputTasks Task[]    @relation("OutputFile")

  @@index([userId, createdAt])
  @@map("files")
}

enum FileType {
  INPUT   // 用户上传的原始文件
  OUTPUT  // 处理后的输出文件
}

// ============ 工具配置表 ============
model Tool {
  id          String       @id // 如 'pdf-to-word'
  name        String       // 如 'PDF 转 Word'
  description String
  category    ToolCategory
  pointsCost  Int          @map("points_cost")
  vipDiscount Float        @default(1.0) @map("vip_discount") // VIP 折扣系数
  maxFileSize Int          @map("max_file_size") // 最大文件大小（字节）
  inputFormats String[]    @map("input_formats") // 支持的输入格式
  outputFormats String[]   @map("output_formats") // 支持的输出格式
  enabled     Boolean      @default(true)
  priority    Int          @default(0) // 排序优先级
  createdAt   DateTime     @default(now()) @map("created_at")
  updatedAt   DateTime     @updatedAt @map("updated_at")

  tasks Task[]

  @@map("tools")
}
```

## 6. 文件处理流程设计

### 6.1 同步处理 vs 异步处理策略

| 处理类型 | 适用场景 | 超时时间 | 说明 |
|---------|---------|---------|------|
| 同步处理 | 图片格式转换、图片压缩、小文件处理 | < 30秒 | 直接在请求中处理，响应等待 |
| 异步处理 | 音视频处理、大文件、文档转换 | 最长 10 分钟 | 入队 BullMQ，前端轮询状态 |

**判断逻辑**：
```typescript
function shouldProcessSync(tool: Tool, file: File): boolean {
  // 图片类且文件小于 20MB → 同步
  if (tool.category === 'IMAGE' && file.size < 20 * 1024 * 1024) return true;
  // 其他一律异步
  return false;
}
```

### 6.2 BullMQ 队列设计

```typescript

const queues = {
```

### 6.3 Worker 实现示例

```typescript
// workers/document.worker.ts
import { Worker, Job } from 'bullmq';
import { convertPdfToWord } from '../processors/document/pdfConvert';
import { prisma } from '../config/database';

const documentWorker = new Worker('file-document', async (job: Job) => {
  const { taskId, toolId, inputFilePath, outputDir, params } = job.data;

  // 更新状态为处理中
  await prisma.task.update({
    where: { id: taskId },
    data: { status: 'PROCESSING', startedAt: new Date() },
  });

  try {
    // 根据 toolId 调用对应处理器
    let outputPath: string;
    switch (toolId) {
      case 'pdf-to-word':
        outputPath = await convertPdfToWord(inputFilePath, outputDir, params);
        break;
      // ... 其他工具
    }

    // 保存输出文件记录
    const outputFile = await prisma.file.create({
      data: {
        userId: job.data.userId,
        originalName: path.basename(outputPath),
        storedName: path.basename(outputPath),
        mimeType: getMimeType(outputPath),
        size: fs.statSync(outputPath).size,
        storagePath: outputPath,
        fileType: 'OUTPUT',
        expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000), // 24小时后过期
      },
    });

    // 更新任务状态为完成
    await prisma.task.update({
      where: { id: taskId },
      data: {
        status: 'COMPLETED',
        outputFileId: outputFile.id,
        completedAt: new Date(),
        progress: 100,
      },
    });

  } catch (error) {
    // 更新任务状态为失败
    await prisma.task.update({
      where: { id: taskId },
      data: {
        status: 'FAILED',
        errorMessage: error.message,
        completedAt: new Date(),
      },
    });

    // 退还积分
    await refundPoints(job.data.userId, job.data.pointsConsumed, taskId);

    throw error; // 让 BullMQ 记录失败
  }
}, {
  concurrency: 3, // 同时处理3个文档任务
  connection: redisConfig,
});
```

```typescript
const WECHAT_AUTH_URL = 'https://open.weixin.qq.com/connect/qrconnect';

const WECHAT_USERINFO_URL = 'https://api.weixin.qq.com/sns/userinfo';

export function getAuthUrl(state: string): string {
  return `${WECHAT_AUTH_URL}?appid=${config.wechat.appId}` +
    `&redirect_uri=${encodeURIComponent(config.wechat.redirectUri)}` +
    `&response_type=code&scope=snsapi_login&state=${state}` +
    `#wechat_redirect`;
}

export async function getAccessToken(code: string) {
  const res = await axios.get(WECHAT_TOKEN_URL, {
    params: {
      appid: config.wechat.appId,
      secret: config.wechat.appSecret,
      code,
      grant_type: 'authorization_code',
    },
  });
  return res.data; // { access_token, openid, unionid, expires_in }
}

export async function getUserInfo(accessToken: string, openid: string) {
  const res = await axios.get(WECHAT_USERINFO_URL, {

}
```

## 9. 安全策略

### 9.1 JWT 认证机制

**双 Token 机制**：
| Token | 有效期 | 存储位置 | 说明 |
|-------|--------|---------|------|
| Access Token | 2 小时 | 前端内存 / sessionStorage | 接口认证 |
| Refresh Token | 30 天 | httpOnly Cookie / localStorage | 刷新 Access Token |

```typescript
// utils/jwt.ts
import jwt from 'jsonwebtoken';

const ACCESS_TOKEN_SECRET = process.env.JWT_ACCESS_SECRET!;
const REFRESH_TOKEN_SECRET = process.env.JWT_REFRESH_SECRET!;

export function signAccessToken(userId: string): string {
  return jwt.sign({ userId }, ACCESS_TOKEN_SECRET, { expiresIn: '2h' });
}

export function signRefreshToken(userId: string): string {
  return jwt.sign({ userId }, REFRESH_TOKEN_SECRET, { expiresIn: '30d' });
}

export function verifyAccessToken(token: string): { userId: string } {
  return jwt.verify(token, ACCESS_TOKEN_SECRET) as { userId: string };
}

// Refresh Token 黑名单（Redis 存储，用于登出时使其失效）
export async function blacklistRefreshToken(userId: string, token: string) {
  const ttl = 30 * 24 * 60 * 60; // 30天
  await redis.set(`refresh:blacklist:${token}`, userId, 'EX', ttl);
}
```

### 9.2 文件上传安全

```typescript
// middleware/upload.ts
import multer from 'multer';
import path from 'path';
import { v4 as uuid } from 'uuid';

// 允许的 MIME 类型白名单
const ALLOWED_MIME_TYPES = new Set([
  // 文档
  'application/pdf',
  'application/msword',
  'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
  'application/vnd.ms-excel',
  'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
  'application/vnd.ms-powerpoint',
  'application/vnd.openxmlformats-officedocument.presentationml.presentation',
  'text/markdown',
  // 图片
  'image/jpeg', 'image/png', 'image/webp', 'image/svg+xml',
  'image/bmp', 'image/heic', 'image/gif',
  // 音频
  'audio/mpeg', 'audio/wav', 'audio/ogg', 'audio/flac',
  'audio/aac', 'audio/mp4',
  // 视频
  'video/mp4', 'video/avi', 'video/quicktime', 'video/x-matroska',
  'video/webm', 'video/x-flv',
]);

const upload = multer({
  storage: multer.diskStorage({
    destination: '/data/uploads/temp',
    filename: (req, file, cb) => {
      const ext = path.extname(file.originalname);
      cb(null, `${uuid()}${ext}`);
    },
  }),
  limits: {
    fileSize: 200 * 1024 * 1024, // 200MB 最大限制
  },
  fileFilter: (req, file, cb) => {
    if (!ALLOWED_MIME_TYPES.has(file.mimetype)) {
      cb(new Error('不支持的文件格式'));
      return;
    }
    cb(null, true);
  },
});
```

**安全措施清单**：
| 措施 | 说明 |
|------|------|
| MIME 类型白名单 | 只允许已知安全的文件类型 |
| 文件扩展名校验 | 双重校验（MIME + 扩展名） |
| 文件大小限制 | 按用户类型和文件类型限制 |
| 文件内容检测 | 读取文件头校验真实类型，防止伪造 MIME |
| 防文件名注入 | 使用 UUID 重命名，不使用原始文件名存储 |
| 隔离存储 | 上传文件存储在非可执行目录 |
| 定期清理 | 定时任务清理过期文件 |
